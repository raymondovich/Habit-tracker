els.cancel.onclick = () => els.modal.classList.remove("show"); els.modal.onclick = e => { if(e.target === els.modal) els.modal.classList.remove("show"); }; els.save.onclick = () => { const name = els.input.value.trim(); if(!name){ els.input.focus(); return; } state.habits.push({ id: Date.now(), name, streak: 0, done: false }); save(); render(); els.modal.classList.remove("show"); }; els.input.onkeydown = e => { if(e.key === "Enter") els.save.click(); }; els.reset.onclick = () => { state.habits.forEach(h => h.done = false); save(); render(); }; render();
